Pridictability
type node struct {
    value int64
    next *node
}

func linkedList(n *node) int64 {
    var total int64
    for n != nil {
        total += n.value 
        n = n.next
    }
    return total 
}

About 70% faster than the linkedList function, despide same spatial locality
